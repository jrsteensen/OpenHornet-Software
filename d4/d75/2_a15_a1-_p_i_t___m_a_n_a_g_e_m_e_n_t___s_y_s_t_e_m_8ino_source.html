115void setup() {
116 // Configure Input Pins
117 pinMode(SIMPWR_MAINT_MODE, INPUT_PULLUP); // Not Yet Implemented
118 pinMode(SIMPWR_PFLT_MODE, INPUT_PULLUP); // Not Yet Implemented
119 pinMode(SIMPWR_DISPON, INPUT_PULLUP); // Latching Switch to power on External Display System
120 pinMode(SIMPWR_MASTERON, INPUT_PULLUP); // Latching Switch to power on Main SimPit Power Supply
121 pinMode(SIMPWR_PCRESET, INPUT_PULLUP); // Momentary Switch to Activate PC Motherboad Reset Switch Function
122 pinMode(SIMPWR_PCPWR, INPUT_PULLUP); // Momentary Switch to Activate PC Motherboad Power Switch Function
123 pinMode(PWR_OK, INPUT); // Feedback circuit from SimPit ATX Power Supply
124
125 // Configure Output Pins
126 pinMode(PS_ON, OUTPUT); // Switch controlled digital output to turn on power supply
127 pinMode(EXT_ACC1, OUTPUT); // Momentary Output for PC Power Switch function (requires external connection to PC motherboard supplied by end-user)
128 pinMode(EXT_ACC2, OUTPUT); // Momentary Output for PC Reset Switch function (requires external connection to PC motherboard supplied by end-user)
129 pinMode(EXT_ACC3, OUTPUT); // Latching Output for Display On Switch Function (requires external connection supplied by end-user)
130 pinMode(EXT_ACC4, OUTPUT); // Not Yet Implemented
131
132 // Setup Pinchange Interupt for Switch Pins
133 PCICR |= 0b00000001; // Enable Pin Change Interupts
134 //PCMSK0 |= 0b00010000; // Select Pin 8 (PS_ON) as interrupt trigger on Interrupt PCINT4"
135 PCMSK0 |= 0b00100000; // Select Pin 9 (PWR_OK) as interrupt trigger on Interrupt PCINT5"
136 //PCMSK0 |= 0b00001000; // Select Pin 14 (Maint_Mode) as interrupt trigger on Interrupt PCINT3"
137 //PCMSK0 |= 0b00000010; // Select Pin 15 (PFLT_MODE) as interrupt trigger on Interrupt PCINT1"
138
139 // Configure Initial Power Supply State Variables prior to first read
140 CurrentSwitchState_SIMPWR_MASTERON = HIGH;
141 CurrentPowerSupplyState = LOW;
142
143 // Setup Internal Controls to Default Off State
144 digitalWrite(PS_ON, HIGH); // Default to off - Power Supply Control
145
146 // Setup External Accessories to Default Off State
147 digitalWrite(EXT_ACC1, LOW); // Default to off - External Accessory #1
148 digitalWrite(EXT_ACC2, LOW); // Default to off - External Accessory #2
149 digitalWrite(EXT_ACC3, LOW); // Default to off - External Accessory #3
150 digitalWrite(EXT_ACC4, LOW); // Default to off - External Accessory #4
151
152 //Serial.begin(9600); //This pipes to the serial monitor
153}

161void loop() {
162 // Read the state of the switches
163 CurrentSwitchState_SIMPWR_MAINT_MODE = digitalRead(SIMPWR_MAINT_MODE); // Get current state of Maintanance Mode Switch, Low = On & High = Off
164 CurrentSwitchState_SIMPWR_PFLT_MODE = digitalRead(SIMPWR_PFLT_MODE); // Get current state of Preflight Mode Switch, Low = On & High = Off
165 CurrentSwitchState_SIMPWR_DISPON = digitalRead(SIMPWR_DISPON); // Get current state of Display On Switch, Low = On & High = Off
166 CurrentSwitchState_SIMPWR_MASTERON = digitalRead(SIMPWR_MASTERON); // Get current state of Master Power Switch, Low = On & High = Off
167 CurrentSwitchState_SIMPWR_PCRESET = digitalRead(SIMPWR_PCRESET); // Get current state of PC Reset Switch, Low = On & High = Off
168 CurrentSwitchState_SIMPWR_PCPWR = digitalRead(SIMPWR_PCPWR); // Get current state of PC Power Switch, Low = On & High = Off
169 CurrentPowerSupplyState = digitalRead(PWR_OK); // Get Current State of Power Supply Status, High = "Power Supply Good!" & Low = "Power Supply Not Ready!"
170
171 delay(20); // Switch Debounce - 20ms

222void Display() {
223 if (CurrentSwitchState_SIMPWR_DISPON != LastSwitchState_SIMPWR_DISPON) {
224 if (CurrentSwitchState_SIMPWR_DISPON == LOW) {
225 digitalWrite(EXT_ACC3, HIGH); // Turn on External Acccessory Port (Latching Switch for External display Subsystem)
226 } else {
227 digitalWrite(EXT_ACC3, LOW); // Turn off External Acccessory Port (Latching Switch for External display Subsystem)
228 }
229 LastSwitchState_SIMPWR_DISPON = CurrentSwitchState_SIMPWR_DISPON;
230 }
231}
